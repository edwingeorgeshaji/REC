## 1. Design direction and priorities

The site pairs a photographic botanical entrance with a restrained technical portfolio. Pink lilies, forest greens, pale sage, handwritten personal lettering, editorial serif headings, and fine dividers establish its identity. Technical SVG illustrations and toolkit motion belong within this quiet visual system. Preserve generous whitespace and clear hierarchy before adding effects.

The home page is a curated introduction, not a complete database. Detailed projects, roles, certificates, biography, and toolkit belong on their dedicated pages. Recruiters should find work, experience, education, contact, and the resume immediately. Never invent achievements, measurements, certificates, testimonials, or proficiency claims. Distinguish illustrative project diagrams and certificate placeholders from real evidence.

Do not restore superseded experiments: blueberry imagery, R.E.C branding, Guido subtitle, custom cursors, a header progress bar, a mobile hamburger, an animated decorative scene at the top of the toolkit subpage, or a contact form. The hero photograph stays still. Company logos are transparent; toolkit logos have intentional discs. Use a short hyphen in titles and date ranges, not an en/em dash. Directional arrow icons are separate visual elements.

const projectRoot=fileURLToPath(new URL('../',import.meta.url));
const root=path.join(projectRoot,'dist');
const production=process.argv.includes('--production');
const build=()=>{
  const result=spawnSync(process.execPath,[path.join(projectRoot,'scripts/build.mjs')],{stdio:'inherit'});
  if(result.status!==0) throw new Error('Build failed');
};
if(!production) build();
await stat(path.join(root,'index.html'));
const types={'.html':'text/html; charset=utf-8','.css':'text/css; charset=utf-8','.js':'application/javascript; charset=utf-8','.mjs':'application/javascript; charset=utf-8','.svg':'image/svg+xml','.pdf':'application/pdf','.woff2':'font/woff2','.otf':'font/otf','.webp':'image/webp','.jpg':'image/jpeg','.xml':'application/xml','.txt':'text/plain; charset=utf-8'};
const server=http.createServer(async(req,res)=>{
  try{
    const pathname=decodeURIComponent(new URL(req.url,'http://localhost').pathname);
    let file=path.resolve(root,'.'+pathname);
    if(file!==root&&!file.startsWith(root+path.sep)){res.writeHead(403);res.end();return;}
    if((await stat(file)).isDirectory()){
      if(!pathname.endsWith('/')){res.writeHead(308,{Location:pathname+'/'});res.end();return;}
      file=path.join(file,'index.html');
    }
    const data=await readFile(file);
    if(pathname==='/resume.pdf')res.setHeader('Content-Disposition','inline; filename="Resume - Edwin George Shaji.pdf"');
    res.writeHead(200,{'Content-Type':types[path.extname(file)]||'application/octet-stream','Cache-Control':'no-store','X-Content-Type-Options':'nosniff'});
    res.end(req.method==='HEAD'?undefined:data);
  }catch{
    res.writeHead(404,{'Content-Type':'text/html; charset=utf-8'});
    res.end(await readFile(path.join(root,'404.html')));
  }
});
server.listen(Number(process.env.PORT||4173),'127.0.0.1',()=>console.log(`Local: http://127.0.0.1:${server.address().port}`));
if(!production){
  let timer;
  for(const directory of ['src','public'])watch(path.join(projectRoot,directory),{recursive:true},()=>{
    clearTimeout(timer);
    timer=setTimeout(()=>{try{build();}catch(error){console.error(error.message);}},160);
  });
  console.log('Watching src/ and public/. Refresh the browser after changes.');
}
